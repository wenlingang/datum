## Review Focus

1. **非 UTF-8 / 控制字符 / emoji 字符串**:文档里的字符串含 `\u0001`、换行、emoji、`"` 时,打开编辑器再原样保存不得改变内容 → Task 4 的往返测试覆盖。
2. **结果排序后错位**:查询 tab 点表头做客户端排序后,"编辑文档 / 删除文档"必须作用于点中的那一篇,而不是排序前同位置的那一篇 → Task 3 的 `sort_order` 测试 + Task 11 的 `permute` 测试。
3. **连接串密码含特殊字符**:密码里有 `@ : / % #` 时,去密码 / 拼回后必须能连上,配置文件里不能出现密码 → Task 5 的 uri 测试。
4. **编辑器里删掉了 `_id` 或改了 `_id`**:删掉时自动补回原值;改了时拒绝保存,绝不能变成"插入一篇新文档" → Task 12 的 `prepare_replacement` 测试。
5. **含不支持类型的文档**(`Code`、`Symbol`、`Undefined`、`DbPointer`):可以显示,编辑器解析失败时禁止保存,不能把字段静默改成字符串 → Task 4 的 `unsupported_types_do_not_round_trip_silently` 测试。

```rust
pub enum DbKind {
    Pg,
    Mysql,
    Redis,
    Sqlite,
    Mongo,
}

impl DbKind {
    pub const ALL: [DbKind; 5] = [
        DbKind::Pg,
        DbKind::Mysql,
        DbKind::Sqlite,
        DbKind::Redis,
        DbKind::Mongo,
    ];

    /// 展示名(状态栏、类型选择)。
    pub fn label(&self) -> &'static str {
        match self {
            DbKind::Pg => "PostgreSQL",
            DbKind::Mysql => "MySQL",
            DbKind::Sqlite => "SQLite",
            DbKind::Redis => "Redis",
            DbKind::Mongo => "MongoDB",
        }
    }

    /// 连接串预览的 scheme。
    pub fn scheme(&self) -> &'static str {
        match self {
            DbKind::Pg => "postgresql",
            DbKind::Mysql => "mysql",
            DbKind::Sqlite => "sqlite",
            DbKind::Redis => "redis",
            DbKind::Mongo => "mongodb",
        }
    }

    /// SQL 编辑器能力:关键字大写、SQL 补全、AI 对话、导入 / 导出 SQL、ERD。
    pub fn supports_sql(&self) -> bool {
        matches!(self, DbKind::Pg | DbKind::Mysql | DbKind::Sqlite)

    /// 是否加载列 schema(补全与 AI 上下文)。
    pub fn has_columns(&self) -> bool {
        self.supports_sql()

    /// 是否提供文档级的编辑 / 新增 / 删除。
    pub fn supports_doc_edit(&self) -> bool {
        matches!(self, DbKind::Mongo)
    }
}
```

```rust
/// 按某列排序后的行下标(稳定排序)。NULL 始终排在最后,与方向无关。
pub fn sort_order(rows: &[Vec<Cell>], col_ix: usize, descending: bool) -> Vec<usize> {
    let mut order: Vec<usize> = (0..rows.len()).collect();
    order.sort_by(|&ia, &ib| match (&rows[ia][col_ix], &rows[ib][col_ix]) {
        (None, None) => Ordering::Equal,
        (None, _) => Ordering::Greater,
        (_, None) => Ordering::Less,
        (Some(a), Some(b)) => {
            let o = compare_cells(a, b);
            if descending { o.reverse() } else { o }
        }
    });
    order
}

/// 按某列对行排序(返回新数组)。NULL 始终排在最后,与方向无关。
pub fn sort_rows(rows: &[Vec<Cell>], col_ix: usize, descending: bool) -> Vec<Vec<Cell>> {
    sort_order(rows, col_ix, descending)
        .into_iter()
        .map(|i| rows[i].clone())
        .collect()
}
```

```rust
//! mongosh 语法子集解析(纯函数,不连网):宽松 JSON → BSON,`db.<coll>.<method>(…)` → [`MongoOp`]。
//!
//! 只接受白名单里的语法,不认识的一律报错并附行列。这样读写判断与风险审查
//! 可以完全基于解析结果,不存在"分析器漏看、服务端照跑"的写法。

use base64::Engine as _;
use mongodb::bson::{
    self, Binary, Bson, Decimal128, Document, Regex, Timestamp, oid::ObjectId, spec::BinarySubtype,
};
use rust_i18n::t;

#[derive(Debug, Clone, PartialEq)]
pub struct ParseError {
    pub line: usize,
    pub col: usize,
    pub message: String,
}

impl ParseError {
    /// 一行可读文案:`第 3 行第 6 列:…`。
    pub fn display(&self) -> String {
        t!("core.mongo.parse.at", line = self.line, col = self.col, message = self.message).to_string()
    }
}

/// 字节偏移 → 1 起的行列(列按字符计)。
fn err_at(src: &str, pos: usize, message: impl Into<String>) -> ParseError {
    let before = &src[..pos.min(src.len())];
    let line = before.matches('\n').count() + 1;
    let col = before.rsplit('\n').next().unwrap_or("").chars().count() + 1;
    ParseError { line, col, message: message.into() }
}

pub(crate) struct P<'a> {
    pub(crate) src: &'a str,
    pub(crate) pos: usize,
}

impl<'a> P<'a> {
    pub(crate) fn new(src: &'a str) -> Self {
        Self { src, pos: 0 }
    }
    fn rest(&self) -> &'a str {
        &self.src[self.pos..]
    }
    fn peek(&self) -> Option<char> {
        self.rest().chars().next()
    }
    fn bump(&mut self) -> Option<char> {
        let c = self.peek()?;
        self.pos += c.len_utf8();
        Some(c)
    }
    pub(crate) fn err(&self, msg: impl Into<String>) -> ParseError {
        err_at(self.src, self.pos, msg)
    }
    pub(crate) fn err_at(&self, pos: usize, msg: impl Into<String>) -> ParseError {
        err_at(self.src, pos, msg)
    }
    pub(crate) fn at_end(&self) -> bool {
        self.pos >= self.src.len()
    }

    /// 跳过空白与 `//`、`/* */` 注释。
    pub(crate) fn ws(&mut self) {
        loop {
            let r = self.rest();
            let t = r.trim_start();
            self.pos += r.len() - t.len();
            if t.starts_with("//") {
                self.pos += t.find('\n').unwrap_or(t.len());
            } else if let Some(body) = t.strip_prefix("/*") {
                self.pos += body.find("*/").map_or(t.len(), |i| i + 4);
            } else {
                break;
            }
        }
    }

    /// 匹配就前进;不匹配时位置不变(连空白也不吞,供语句分隔判断用)。
    pub(crate) fn eat(&mut self, s: &str) -> bool {
        let save = self.pos;
        self.ws();
        if self.rest().starts_with(s) {
            self.pos += s.len();
            true
        } else {
            self.pos = save;
            false
        }
    }

    pub(crate) fn expect(&mut self, s: &str) -> Result<(), ParseError> {
        if self.eat(s) {
            Ok(())
        } else {
            self.ws();
            Err(self.err(t!("core.mongo.parse.expected", what = s)))
        }
    }

    /// 标识符:字母 / 数字 / `_` / `$`,不能以数字开头。
    pub(crate) fn ident(&mut self) -> Option<&'a str> {
        let save = self.pos;
        self.ws();
        let r = self.rest();
        let end = r
            .char_indices()
            .find(|(_, c)| !(c.is_alphanumeric() || *c == '_' || *c == '$'))
            .map_or(r.len(), |(i, _)| i);
        if end == 0 || r.starts_with(|c: char| c.is_ascii_digit()) {
            self.pos = save;
            return None;
        }
        self.pos += end;
        Some(&r[..end])
    }

    pub(crate) fn value(&mut self) -> Result<Bson, ParseError> {
        self.ws();
        match self.peek() {
            Some('{') => self.object().map(Bson::Document),
            Some('[') => self.array().map(Bson::Array),
            Some('"') | Some('\'') => self.string().map(Bson::String),
            Some('/') => self.regex(),
            Some(c) if c == '-' || c == '+' || c == '.' || c.is_ascii_digit() => self.number(),
            Some(_) => self.word_value(),
            None => Err(self.err(t!("core.mongo.parse.unexpected_end"))),
        }
    }

    fn object(&mut self) -> Result<Document, ParseError> {
        self.expect("{")?;
        let mut doc = Document::new();
        loop {
            if self.eat("}") {
                return Ok(doc);
            }
            let key = self.key()?;
            self.expect(":")?;
            let v = self.value()?;
            doc.insert(key, v);
            if !self.eat(",") {
                self.expect("}")?;
                return Ok(doc);
            }
        }
    }

    fn key(&mut self) -> Result<String, ParseError> {
        self.ws();
        match self.peek() {
            Some('"') | Some('\'') => self.string(),
            _ => self
                .ident()
                .map(str::to_string)
                .ok_or_else(|| self.err(t!("core.mongo.parse.expected_key"))),
        }
    }

    pub(crate) fn array(&mut self) -> Result<Vec<Bson>, ParseError> {
        self.expect("[")?;
        let mut out = vec![];
        loop {
            if self.eat("]") {
                return Ok(out);
            }
            out.push(self.value()?);
            if !self.eat(",") {
                self.expect("]")?;
                return Ok(out);
            }
        }
    }

    pub(crate) fn string(&mut self) -> Result<String, ParseError> {
        self.ws();
        let start = self.pos;
        let Some(q) = self.bump().filter(|c| *c == '"' || *c == '\'') else {
            return Err(self.err_at(start, t!("core.mongo.parse.expected_string")));
        };
        let mut s = String::new();
        loop {
            match self.bump() {
                None => return Err(self.err_at(start, t!("core.mongo.parse.unterminated_string"))),
                Some(c) if c == q => return Ok(s),
                Some('\\') => {
                    let esc_pos = self.pos - 1;
                    match self.bump() {
                        Some('n') => s.push('\n'),
                        Some('t') => s.push('\t'),
                        Some('r') => s.push('\r'),
                        Some('b') => s.push('\u{8}'),
                        Some('f') => s.push('\u{c}'),
                        Some('0') => s.push('\0'),
                        Some('u') => s.push(self.unicode_escape(esc_pos)?),
                        Some(c) => s.push(c),
                        None => return Err(self.err_at(start, t!("core.mongo.parse.unterminated_string"))),
                    }
                }
                Some(c) => s.push(c),
            }
        }
    }

    /// `\uXXXX`(已吞掉 `\u`),处理代理对。
    fn unicode_escape(&mut self, esc_pos: usize) -> Result<char, ParseError> {
        let bad = |p: &Self| p.err_at(esc_pos, t!("core.mongo.parse.bad_escape"));
        let hex4 = |p: &mut Self| -> Option<u32> {
            let h = p.rest().get(..4)?;
            let n = u32::from_str_radix(h, 16).ok()?;
            p.pos += 4;
            Some(n)
        };
        let hi = hex4(self).ok_or_else(|| bad(self))?;
        if (0xD800..0xDC00).contains(&hi) {
            if !self.rest().starts_with("\\u") {
                return Err(bad(self));
            }
            self.pos += 2;
            let lo = hex4(self).ok_or_else(|| bad(self))?;
            if !(0xDC00..0xE000).contains(&lo) {
                return Err(bad(self));
            }
            let c = 0x10000 + ((hi - 0xD800) << 10) + (lo - 0xDC00);
            return char::from_u32(c).ok_or_else(|| bad(self));
        }
        char::from_u32(hi).ok_or_else(|| bad(self))
    }

    /// 数字 token 的原文(含符号、小数点、指数,以及 Infinity)。
    fn number_text(&mut self) -> &'a str {
        self.ws();
        let r = self.rest();
        let b = r.as_bytes();
        let end = r
            .char_indices()
            .find(|&(i, c)| {
                let sign_ok = (c == '-' || c == '+') && (i == 0 || matches!(b[i - 1], b'e' | b'E'));
                !(c.is_ascii_alphanumeric() || c == '.' || sign_ok)
            })
            .map_or(r.len(), |(i, _)| i);
        self.pos += end;
        &r[..end]
    }

    fn number(&mut self) -> Result<Bson, ParseError> {
        self.ws();
        let start = self.pos;
        let text = self.number_text();
        parse_number(text)
            .ok_or_else(|| self.err_at(start, t!("core.mongo.parse.bad_number", text = text)))
    }

    fn regex(&mut self) -> Result<Bson, ParseError> {
        let start = self.pos;
        self.bump(); // '/'
        let mut pattern = String::new();
        let mut in_class = false;
        loop {
            match self.bump() {
                None | Some('\n') => {
                    return Err(self.err_at(start, t!("core.mongo.parse.unterminated_regex")));
                }
                Some('\\') => {
                    pattern.push('\\');
                    if let Some(c) = self.bump() {
                        pattern.push(c);
                    }
                }
                Some('[') => {
                    in_class = true;
                    pattern.push('[');
                }
                Some(']') => {
                    in_class = false;
                    pattern.push(']');
                }
                Some('/') if !in_class => break,
                Some(c) => pattern.push(c),
            }
        }
        let r = self.rest();
        let end = r.find(|c: char| !c.is_ascii_alphabetic()).unwrap_or(r.len());
        let options = sorted_flags(&r[..end]);
        self.pos += end;
        Ok(Bson::RegularExpression(Regex { pattern, options }))
    }

    /// 字符串参数或裸数字参数的原文(`NumberLong(5)` 与 `NumberLong("5")` 都可以)。
    fn num_or_str_arg(&mut self) -> Result<String, ParseError> {
        self.ws();
        match self.peek() {
            Some('"') | Some('\'') => self.string(),
            _ => Ok(self.number_text().to_string()),
        }
    }

    fn int_arg(&mut self) -> Result<i64, ParseError> {
        self.ws();
        let pos = self.pos;
        let s = self.num_or_str_arg()?;
        s.trim()
            .parse::<i64>()
            .map_err(|_| self.err_at(pos, t!("core.mongo.parse.bad_number", text = s)))
    }

    fn word_value(&mut self) -> Result<Bson, ParseError> {
        self.ws();
        let start = self.pos;
        let Some(word) = self.ident() else {
            let ch = self.peek().unwrap_or(' ');
            return Err(self.err(t!("core.mongo.parse.unexpected", ch = ch)));
        };
        match word {
            "true" => return Ok(Bson::Boolean(true)),
            "false" => return Ok(Bson::Boolean(false)),
            "null" => return Ok(Bson::Null),
            "NaN" => return Ok(Bson::Double(f64::NAN)),
            "Infinity" => return Ok(Bson::Double(f64::INFINITY)),
            "new" => {
                if self.ident() != Some("Date") {
                    return Err(self.err_at(start, t!("core.mongo.parse.unsupported_ctor", name = "new")));
                }
                return self.date_ctor();
            }
            "ISODate" => return self.date_ctor(),
            _ => {}
        }
        self.expect("(")?;
        self.ws();
        let arg_pos = self.pos;
        let bad = |p: &Self| p.err_at(arg_pos, t!("core.mongo.parse.bad_ctor_arg", name = word));
        let v = match word {
            "ObjectId" => {
                let s = self.string()?;
                Bson::ObjectId(ObjectId::parse_str(&s).map_err(|_| bad(self))?)
            }
            "NumberInt" | "Int32" => {
                Bson::Int32(i32::try_from(self.int_arg()?).map_err(|_| bad(self))?)
            }
            "NumberLong" | "Long" => Bson::Int64(self.int_arg()?),
            "Double" => {
                let s = self.num_or_str_arg()?;
                Bson::Double(s.trim().parse::<f64>().map_err(|_| bad(self))?)
            }
            "NumberDecimal" | "Decimal128" => {
                let s = self.num_or_str_arg()?;
                Bson::Decimal128(s.trim().parse::<Decimal128>().map_err(|_| bad(self))?)
            }
            "UUID" => {
                let s = self.string()?;
                let u = bson::Uuid::parse_str(&s).map_err(|_| bad(self))?;
                Bson::Binary(Binary { subtype: BinarySubtype::Uuid, bytes: u.bytes().to_vec() })
            }
            "BinData" => {
                let sub = self.int_arg()?;
                let sub = u8::try_from(sub).map_err(|_| bad(self))?;
                self.expect(",")?;
                let b64 = self.string()?;
                let bytes = base64::engine::general_purpose::STANDARD
                    .decode(b64.as_bytes())
                    .map_err(|_| bad(self))?;
                Bson::Binary(Binary { subtype: BinarySubtype::from(sub), bytes })
            }
            "Timestamp" => {
                let time = u32::try_from(self.int_arg()?).map_err(|_| bad(self))?;
                self.expect(",")?;
                let increment = u32::try_from(self.int_arg()?).map_err(|_| bad(self))?;
                Bson::Timestamp(Timestamp { time, increment })
            }
            "RegExp" => {
                let pattern = self.string()?;
                let options = if self.eat(",") { sorted_flags(&self.string()?) } else { String::new() };
                Bson::RegularExpression(Regex { pattern, options })
            }
            "MinKey" => Bson::MinKey,
            "MaxKey" => Bson::MaxKey,
            _ => return Err(self.err_at(start, t!("core.mongo.parse.unsupported_ctor", name = word))),
        };
        self.expect(")")?;
        Ok(v)
    }

    /// `ISODate(...)` / `new Date(...)`:无参为当前时间,字符串按 ISO-8601,数字为毫秒。
    fn date_ctor(&mut self) -> Result<Bson, ParseError> {
        self.expect("(")?;
        if self.eat(")") {
            return Ok(Bson::DateTime(bson::DateTime::now()));
        }
        self.ws();
        let pos = self.pos;
        let ms = match self.peek() {
            Some('"') | Some('\'') => {
                let s = self.string()?;
                parse_date_millis(&s).ok_or_else(|| self.err_at(pos, t!("core.mongo.parse.bad_date", text = s)))?
            }
            _ => self.int_arg()?,
        };
        self.expect(")")?;
        Ok(Bson::DateTime(bson::DateTime::from_millis(ms)))
    }
}

/// 正则 flags 排序后输出,和服务端的规范化保持一致。
fn sorted_flags(s: &str) -> String {
    let mut f: Vec<char> = s.chars().collect();
    f.sort_unstable();
    f.into_iter().collect()
}

/// 整数在 i32 范围内为 Int32、超出为 Int64;带小数点或指数为 Double。
/// 注意:mongosh 会把所有裸数字当 Double,这里保留整数类型,写入时更不容易意外变成浮点。
fn parse_number(text: &str) -> Option<Bson> {
    match text {
        "Infinity" | "+Infinity" => return Some(Bson::Double(f64::INFINITY)),
        "-Infinity" => return Some(Bson::Double(f64::NEG_INFINITY)),
        _ => {}
    }
    if !text.contains(['.', 'e', 'E']) {
        if let Ok(n) = text.parse::<i64>() {
            return Some(i32::try_from(n).map_or(Bson::Int64(n), Bson::Int32));
        }
    }
    text.parse::<f64>().ok().filter(|f| f.is_finite()).map(Bson::Double)
}

fn parse_date_millis(s: &str) -> Option<i64> {
    use chrono::{DateTime, NaiveDate, NaiveDateTime};
    let s = s.trim();
    if let Ok(d) = DateTime::parse_from_rfc3339(s) {
        return Some(d.timestamp_millis());
    }
    if let Ok(d) = NaiveDateTime::parse_from_str(s, "%Y-%m-%dT%H:%M:%S%.f") {
        return Some(d.and_utc().timestamp_millis());
    }
    NaiveDate::parse_from_str(s, "%Y-%m-%d")
        .ok()
        .and_then(|d| d.and_hms_opt(0, 0, 0))
        .map(|d| d.and_utc().timestamp_millis())
}

/// 解析一个完整的值;后面不能再有多余内容。
pub fn parse_value(src: &str) -> Result<Bson, ParseError> {
    let mut p = P::new(src);
    let v = p.value()?;
    p.ws();
    if !p.at_end() {
        return Err(p.err(t!("core.mongo.parse.trailing")));
    }
    Ok(v)
}

/// 解析顶层为对象的文本(过滤栏、文档编辑器)。空串或纯空白视为 `{}`。
```

```rust
//! BSON → 展示:mongosh 风格文本(JSON 视图与文档编辑器)与单元格文本(表格视图)。
//!
//! `to_shell` 的输出恰好是 [`parse_value`](super::parse::parse_value) 能解析的语法,
//! 且类型标注齐全,保证"打开编辑器 → 原样保存"不改变任何 BSON 类型。

use base64::Engine as _;
use mongodb::bson::{Bson, Document, spec::BinarySubtype};

pub fn to_shell(v: &Bson, pretty: bool) -> String {
    let mut out = String::new();
    write(&mut out, v, pretty, 0);
    out
}

fn json_str(s: &str) -> String {
    serde_json::to_string(s).unwrap_or_else(|_| format!("{s:?}"))
}

/// 能作为裸 key 输出的字段名(否则加引号)。
fn bare_key(k: &str) -> bool {
    !k.is_empty()
        && !k.starts_with(|c: char| c.is_ascii_digit())
        && k.chars().all(|c| c.is_alphanumeric() || c == '_' || c == '$')
}

fn newline(out: &mut String, pretty: bool, indent: usize) {
    if pretty {
        out.push('\n');
        out.push_str(&"  ".repeat(indent));
    }
}

fn write_doc(out: &mut String, d: &Document, pretty: bool, indent: usize) {
    if d.is_empty() {
        out.push_str("{}");
        return;
    }
    out.push('{');
    for (i, (k, v)) in d.iter().enumerate() {
        if i > 0 {
            out.push(',');
        }
        if pretty { newline(out, true, indent + 1) } else { out.push(' ') }
        out.push_str(&if bare_key(k) { k.clone() } else { json_str(k) });
        out.push_str(": ");
        write(out, v, pretty, indent + 1);
    }
    if pretty { newline(out, true, indent) } else { out.push(' ') }
    out.push('}');
}

fn write(out: &mut String, v: &Bson, pretty: bool, indent: usize) {
    match v {
        Bson::Document(d) => write_doc(out, d, pretty, indent),
        Bson::Array(a) => {
            if a.is_empty() {
                out.push_str("[]");
                return;
            }
            out.push('[');
            for (i, item) in a.iter().enumerate() {
                if i > 0 {
                    out.push(',');
                    if !pretty {
                        out.push(' ');
                    }
                }
                newline(out, pretty, indent + 1);
                write(out, item, pretty, indent + 1);
            }
            newline(out, pretty, indent);
            out.push(']');
        }
        other => out.push_str(&scalar_shell(other)),
    }
}

fn fmt_double(f: f64) -> String {
    if f.is_nan() {
        "NaN".into()
    } else if f.is_infinite() {
        if f > 0. { "Infinity".into() } else { "-Infinity".into() }
    } else if f.fract() == 0. && f.abs() < 1e15 {
        format!("{f:.1}")
    } else {
        // Debug 输出最短且可精确回读的表示(如 0.1、1e20)
        format!("{f:?}")
    }
}

/// 毫秒 → `2024-01-02T03:04:05.678Z`;年份超出 0–9999 时返回 None。
fn iso_millis(ms: i64) -> Option<String> {
    use chrono::{Datelike as _, SecondsFormat};
    let d = chrono::DateTime::from_timestamp_millis(ms)?;
    (0..=9999)
        .contains(&d.year())
        .then(|| d.to_rfc3339_opts(SecondsFormat::Millis, true))
}

fn scalar_shell(v: &Bson) -> String {
    match v {
        Bson::String(s) => json_str(s),
        Bson::Boolean(b) => b.to_string(),
        Bson::Null => "null".into(),
        Bson::Int32(n) => n.to_string(),
        Bson::Int64(n) => format!("NumberLong({n})"),
        Bson::Double(f) => fmt_double(*f),
        Bson::Decimal128(d) => format!("NumberDecimal(\"{d}\")"),
        Bson::ObjectId(o) => format!("ObjectId(\"{o}\")"),
        Bson::DateTime(dt) => {
            let ms = dt.timestamp_millis();
            match iso_millis(ms) {
                Some(s) => format!("ISODate(\"{s}\")"),
                None => format!("new Date({ms})"),
            }
        }
        Bson::Binary(b) if b.subtype == BinarySubtype::Uuid && b.bytes.len() == 16 => {
            let u = mongodb::bson::Uuid::from_bytes(b.bytes.clone().try_into().unwrap_or([0; 16]));
            format!("UUID(\"{u}\")")
        }
        Bson::Binary(b) => format!(
            "BinData({}, \"{}\")",
            u8::from(b.subtype),
            base64::engine::general_purpose::STANDARD.encode(&b.bytes)
        ),
        Bson::RegularExpression(r) if r.pattern.contains(['/', '\n']) => {
            format!("RegExp({}, {})", json_str(&r.pattern), json_str(&r.options))
        }
        Bson::RegularExpression(r) => format!("/{}/{}", r.pattern, r.options),
        Bson::Timestamp(t) => format!("Timestamp({}, {})", t.time, t.increment),
        Bson::MinKey => "MinKey()".into(),
        Bson::MaxKey => "MaxKey()".into(),
        // Code / Symbol / Undefined / DbPointer:只做展示,解析器不接受,编辑器会拒绝保存
        other => format!("{other}"),
    }
}

/// 表格单元格文本:字符串不加引号,数字不带类型包装,日期为 ISO,嵌套值为紧凑 mongosh 文本。
pub fn cell_text(v: &Bson) -> String {
    match v {
        Bson::String(s) => s.clone(),
        Bson::Int32(n) => n.to_string(),
        Bson::Int64(n) => n.to_string(),
        Bson::Double(f) => fmt_double(*f),
        Bson::Decimal128(d) => d.to_string(),
        Bson::DateTime(dt) => {
            iso_millis(dt.timestamp_millis()).unwrap_or_else(|| dt.timestamp_millis().to_string())
        }
        other => to_shell(other, false),
    }
}
```

```rust
//! 连接串处理(纯函数):拆出密码与库名、按需拼回密码、表单字段拼连接串。
//!
//! 配置文件里只存不含密码、不含库名的连接串;密码走现有的 `credentials`(Local 明文、其余进钥匙串),
//! 库名放 `Connection.database`,这样侧栏切库只改 `database`,不用改连接串。

use crate::core::connection::{Connection, SslMode};

pub struct UriParts {
    /// 不含密码与库名的连接串
    pub sanitized: String,
    pub user: String,
    pub password: Option<String>,
    /// 主机列表原文(`h1:27017,h2` 或 SRV 域名),用于侧栏展示与切库匹配
    pub hosts: String,
    pub database: String,
}

/// RFC 3986 unreserved 之外的字节一律百分号编码。
fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{b:02X}"));
        }
    }
    out
}

fn decode(s: &str) -> Result<String, String> {
    let bytes = s.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'%' {
            let hex = s.get(i + 1..i + 3).ok_or("bad percent-encoding")?;
            out.push(u8::from_str_radix(hex, 16).map_err(|_| "bad percent-encoding")?);
            i += 3;
        } else {
            out.push(bytes[i]);
            i += 1;
        }
    }
    String::from_utf8(out).map_err(|_| "bad percent-encoding".to_string())
}

/// 拆成 (scheme://, userinfo, hosts, path, query)。
fn pieces(input: &str) -> Result<(&str, Option<&str>, &str, &str, &str), String> {
    let scheme = ["mongodb+srv://", "mongodb://"]
        .into_iter()
        .find(|s| input.starts_with(s))
        .ok_or("connection string must start with mongodb:// or mongodb+srv://")?;
    let rest = &input[scheme.len()..];
    let (before_q, query) = rest.split_once('?').unwrap_or((rest, ""));
    let (authority, path) = before_q.split_once('/').unwrap_or((before_q, ""));
    let (userinfo, hosts) = match authority.rsplit_once('@') {
        Some((u, h)) => (Some(u), h),
        None => (None, authority),
    };
    if hosts.is_empty() {
        return Err("connection string has no host".into());
    }
    Ok((scheme, userinfo, hosts, path, query))
}

pub fn split_uri(input: &str) -> Result<UriParts, String> {
    let input = input.trim();
    mongodb::options::ConnectionString::parse(input).map_err(|e| e.to_string())?;
    let (scheme, userinfo, hosts, path, query) = pieces(input)?;
    let (user, password) = match userinfo {
        Some(u) => match u.split_once(':') {
            Some((name, pw)) => (decode(name)?, Some(decode(pw)?)),
            None => (decode(u)?, None),
        },
        None => (String::new(), None),
    };
    let database = decode(path)?;
    let mut params: Vec<String> = query
        .split('&')
        .filter(|p| !p.is_empty())
        .map(str::to_string)
        .collect();
    // 路径里的库名原本兼作认证库;去掉库名后显式写上,避免认证库悄悄变成 admin
    let has_auth_source = params
        .iter()
        .any(|p| p.split('=').next().is_some_and(|k| k.eq_ignore_ascii_case("authSource")));
    if !database.is_empty() && !user.is_empty() && !has_auth_source {
        params.push(format!("authSource={}", encode(&database)));
    }
    let userpart = if user.is_empty() { String::new() } else { format!("{}@", encode(&user)) };
    let query = if params.is_empty() { String::new() } else { format!("?{}", params.join("&")) };
    Ok(UriParts {
        sanitized: format!("{scheme}{userpart}{hosts}/{query}"),
        user,
        password,
        hosts: hosts.to_string(),
        database,
    })
}

/// 把密码拼回已去密码的连接串;没有用户名或密码为空时原样返回。
pub fn with_password(sanitized: &str, password: &str) -> String {
    if password.is_empty() {
        return sanitized.to_string();
    }
    let Ok((scheme, Some(user), _, _, _)) = pieces(sanitized) else {
        return sanitized.to_string();
    };
    let rest = &sanitized[scheme.len() + user.len()..];
    format!("{scheme}{user}:{}{rest}", encode(password))
}

/// 用 `••••` 遮住密码,供表单预览。
pub fn redact(uri: &str) -> String {
    let Ok((scheme, Some(userinfo), _, _, _)) = pieces(uri) else {
        return uri.to_string();
    };
    match userinfo.split_once(':') {
        Some((user, _)) => format!("{scheme}{user}:••••{}", &uri[scheme.len() + userinfo.len()..]),
        None => uri.to_string(),
    }
}

/// 表单模式拼连接串。`Prefer` 返回两个候选:先加密(不校验证书),失败再明文。
/// 单主机表单默认 directConnection,直连副本集成员也能用(与 mongosh 行为一致)。
pub fn form_uris(
    host: &str,
    port: u16,
    user: &str,
    password: &str,
    auth_source: Option<&str>,
    ssl: SslMode,
) -> Vec<String> {
    let auth = if user.is_empty() {
        String::new()
    } else if password.is_empty() {
        format!("{}@", encode(user))
    } else {
        format!("{}:{}@", encode(user), encode(password))
    };
    let mut params = vec!["directConnection=true".to_string()];
    if !user.is_empty() {
        let src = auth_source.filter(|s| !s.trim().is_empty()).unwrap_or("admin");
        params.push(format!("authSource={}", encode(src.trim())));
    }
    let base = format!("mongodb://{auth}{host}:{port}/?{}", params.join("&"));
    let tls = |flags: &str| format!("{base}&{flags}");
    match ssl {
        SslMode::Disable => vec![tls("tls=false")],
        SslMode::Prefer => vec![tls("tls=true&tlsAllowInvalidCertificates=true"), tls("tls=false")],
        SslMode::Require => vec![tls("tls=true&tlsAllowInvalidCertificates=true")],
        SslMode::VerifyFull => vec![tls("tls=true")],
    }
}

/// 已保存连接的候选连接串(按顺序尝试)。
pub fn connect_uris(conn: &Connection, password: &str) -> Vec<String> {
    match &conn.uri {
        Some(u) => vec![with_password(u, password)],
        None => form_uris(&conn.host, conn.port, &conn.user, password, conn.auth_source.as_deref(), conn.ssl_mode),
    }
}
```

```rust
#[derive(Debug, Clone, PartialEq)]
pub enum Update {
    Doc(Document),
    Pipeline(Vec<Document>),
}

#[derive(Debug, Clone, PartialEq)]
pub enum MongoOp {
    ShowDbs,
    ShowCollections,
    Find { coll: String, filter: Document, projection: Option<Document>, sort: Option<Document>, skip: Option<u64>, limit: Option<i64> },
    FindOne { coll: String, filter: Document, projection: Option<Document> },
    Count { coll: String, filter: Document },
    Distinct { coll: String, field: String, filter: Document },
    Aggregate { coll: String, pipeline: Vec<Document> },
    GetIndexes { coll: String },
    InsertOne { coll: String, doc: Document },
    InsertMany { coll: String, docs: Vec<Document> },
    UpdateOne { coll: String, filter: Document, update: Update },
    UpdateMany { coll: String, filter: Document, update: Update },
    ReplaceOne { coll: String, filter: Document, doc: Document },
    DeleteOne { coll: String, filter: Document },
    DeleteMany { coll: String, filter: Document },
}

impl MongoOp {
    /// 写操作:7 个写方法,以及管道中任意位置含 `$out` / `$merge` 的 aggregate。
    pub fn is_write(&self) -> bool {
        match self {
            MongoOp::InsertOne { .. }
            | MongoOp::InsertMany { .. }
            | MongoOp::UpdateOne { .. }
            | MongoOp::UpdateMany { .. }
            | MongoOp::ReplaceOne { .. }
            | MongoOp::DeleteOne { .. }
            | MongoOp::DeleteMany { .. } => true,
            MongoOp::Aggregate { pipeline, .. } => pipeline
                .iter()
                .any(|s| s.contains_key("$out") || s.contains_key("$merge")),
            _ => false,
        }
    }

    pub fn collection(&self) -> Option<&str> {
        match self {
            MongoOp::ShowDbs | MongoOp::ShowCollections => None,
            MongoOp::Find { coll, .. }
            | MongoOp::FindOne { coll, .. }
            | MongoOp::Count { coll, .. }
            | MongoOp::Distinct { coll, .. }
            | MongoOp::Aggregate { coll, .. }
            | MongoOp::GetIndexes { coll }
            | MongoOp::InsertOne { coll, .. }
            | MongoOp::InsertMany { coll, .. }
            | MongoOp::UpdateOne { coll, .. }
            | MongoOp::UpdateMany { coll, .. }
            | MongoOp::ReplaceOne { coll, .. }
            | MongoOp::DeleteOne { coll, .. }
            | MongoOp::DeleteMany { coll, .. } => Some(coll),
        }
    }
}

#[derive(Debug, Clone, PartialEq)]
pub struct Statement {
    /// 语句在原文中的字节范围(风险弹窗展示原文)
    pub range: std::ops::Range<usize>,
    pub op: MongoOp,
}

/// 集合名能作为标识符时写成 `db.name`,否则 `db.getCollection("…")`。
pub fn coll_ref(coll: &str) -> String {
    let ident = !coll.is_empty()
        && !coll.starts_with(|c: char| c.is_ascii_digit())
        && coll.chars().all(|c| c.is_alphanumeric() || c == '_' || c == '$');
    if ident {
        format!("db.{coll}")
    } else {
        format!("db.getCollection({})", serde_json::to_string(coll).unwrap_or_default())
    }
}

/// 解析脚本:语句之间用 `;` 或换行分隔。
pub fn parse_script(src: &str) -> Result<Vec<Statement>, ParseError> {
    let mut p = P::new(src);
    let mut out = vec![];
    loop {
        while p.eat(";") {}
        p.ws();
        if p.at_end() {
            return Ok(out);
        }
        let start = p.pos;
        let op = p.statement()?;
        let end = p.pos;
        out.push(Statement { range: start..end, op });
        p.ws();
        if !p.at_end() && !p.src[p.pos..].starts_with(';') && !src[end..p.pos].contains('\n') {
            return Err(p.err(t!("core.mongo.parse.expected_separator")));
        }
    }
}

/// 方法参数:值 + 起始位置(报错定位用)。
struct Args<'s> {
    src: &'s str,
    method: &'s str,
    call_pos: usize,
    items: Vec<(usize, Bson)>,
}

impl Args<'_> {
    fn err(&self, pos: usize, msg: impl Into<String>) -> ParseError {
        err_at(self.src, pos, msg)
    }
    fn max(&self, n: usize) -> Result<(), ParseError> {
        match self.items.get(n) {
            Some((pos, _)) => Err(self.err(*pos, t!("core.mongo.parse.too_many_args", name = self.method, n = n))),
            None => Ok(()),
        }
    }
    fn doc(&self, i: usize) -> Result<Document, ParseError> {
        match self.items.get(i) {
            Some((_, Bson::Document(d))) => Ok(d.clone()),
            Some((pos, _)) => Err(self.err(*pos, t!("core.mongo.parse.expected_object"))),
            None => Err(self.err(self.call_pos, t!("core.mongo.parse.missing_arg", name = self.method))),
        }
    }
    fn doc_or_empty(&self, i: usize) -> Result<Document, ParseError> {
        if self.items.get(i).is_none() { Ok(Document::new()) } else { self.doc(i) }
    }
    fn opt_doc(&self, i: usize) -> Result<Option<Document>, ParseError> {
        if self.items.get(i).is_none() { Ok(None) } else { self.doc(i).map(Some) }
    }
    fn doc_array(&self, i: usize) -> Result<Vec<Document>, ParseError> {
        let Some((pos, v)) = self.items.get(i) else {
            return Err(self.err(self.call_pos, t!("core.mongo.parse.missing_arg", name = self.method)));
        };
        let Bson::Array(items) = v else {
            return Err(self.err(*pos, t!("core.mongo.parse.expected_doc_array")));
        };
        items
            .iter()
            .map(|b| match b {
                Bson::Document(d) => Ok(d.clone()),
                _ => Err(self.err(*pos, t!("core.mongo.parse.expected_doc_array"))),
            })
            .collect()
    }
    fn string(&self, i: usize) -> Result<String, ParseError> {
        match self.items.get(i) {
            Some((_, Bson::String(s))) => Ok(s.clone()),
            Some((pos, _)) => Err(self.err(*pos, t!("core.mongo.parse.expected_string"))),
            None => Err(self.err(self.call_pos, t!("core.mongo.parse.missing_arg", name = self.method))),
        }
    }
    fn update(&self, i: usize) -> Result<Update, ParseError> {
        match self.items.get(i) {
            Some((_, Bson::Array(_))) => self.doc_array(i).map(Update::Pipeline),
            Some((pos, Bson::Document(d))) => {
                if d.is_empty() || d.keys().any(|k| !k.starts_with('$')) {
                    return Err(self.err(*pos, t!("core.mongo.parse.update_needs_operator")));
                }
                Ok(Update::Doc(d.clone()))
            }
            Some((pos, _)) => Err(self.err(*pos, t!("core.mongo.parse.expected_object"))),
            None => Err(self.err(self.call_pos, t!("core.mongo.parse.missing_arg", name = self.method))),
        }
    }
    fn uint(&self, i: usize) -> Result<i64, ParseError> {
        match self.items.get(i) {
            Some((_, Bson::Int32(n))) if *n >= 0 => Ok(*n as i64),
            Some((_, Bson::Int64(n))) if *n >= 0 => Ok(*n),
            Some((pos, _)) => Err(self.err(*pos, t!("core.mongo.parse.expected_uint"))),
            None => Err(self.err(self.call_pos, t!("core.mongo.parse.missing_arg", name = self.method))),
        }
    }
}

impl P<'_> {
    /// `(` 已吞掉;读到 `)` 为止,允许末尾逗号。
    fn call_args(&mut self) -> Result<Vec<(usize, Bson)>, ParseError> {
        let mut items = vec![];
        loop {
            if self.eat(")") {
                return Ok(items);
            }
            self.ws();
            let pos = self.pos;
            items.push((pos, self.value()?));
            if !self.eat(",") {
                self.expect(")")?;
                return Ok(items);
            }
        }
    }

    pub(crate) fn statement(&mut self) -> Result<MongoOp, ParseError> {
        self.ws();
        let start = self.pos;
        let head = self.ident();
        if head == Some("show") {
            let what_pos = self.pos;
            return match self.ident() {
                Some("dbs") | Some("databases") => Ok(MongoOp::ShowDbs),
                Some("collections") | Some("tables") => Ok(MongoOp::ShowCollections),
                _ => Err(self.err_at(what_pos, t!("core.mongo.parse.unsupported_show"))),
            };
        }
        if head != Some("db") {
            return Err(self.err_at(start, t!("core.mongo.parse.expected_statement")));
        }
        self.expect(".")?;
        self.ws();
        let name_pos = self.pos;
        let coll = match self.ident() {
            Some("getCollection") => {
                self.expect("(")?;
                let s = self.string()?;
                self.expect(")")?;
                s
            }
            Some(name) => name.to_string(),
            None => return Err(self.err_at(name_pos, t!("core.mongo.parse.expected_collection"))),
        };
        self.expect(".")?;
        self.ws();
        let call_pos = self.pos;
        let method = self
            .ident()
            .ok_or_else(|| self.err(t!("core.mongo.parse.expected_method")))?;
        self.expect("(")?;
        let items = self.call_args()?;
        let a = Args { src: self.src, method, call_pos, items };
        let mut op = build_op(&a, coll)?;
        self.chain(&mut op)?;
        Ok(op)
    }

    /// `find()` 之后的 `.sort() .skip() .limit() .projection() .pretty() .toArray()`。
    fn chain(&mut self, op: &mut MongoOp) -> Result<(), ParseError> {
        while self.eat(".") {
            self.ws();
            let call_pos = self.pos;
            let method = self
                .ident()
                .ok_or_else(|| self.err(t!("core.mongo.parse.expected_method")))?;
            self.expect("(")?;
            let items = self.call_args()?;
            let a = Args { src: self.src, method, call_pos, items };
            let MongoOp::Find { sort, skip, limit, projection, .. } = op else {
                return Err(self.err_at(call_pos, t!("core.mongo.parse.chain_only_after_find", name = method)));
            };
            match method {
                "sort" => { a.max(1)?; *sort = Some(a.doc(0)?); }
                "skip" => { a.max(1)?; *skip = Some(a.uint(0)? as u64); }
                "limit" => { a.max(1)?; *limit = Some(a.uint(0)?); }
                "projection" => { a.max(1)?; *projection = Some(a.doc(0)?); }
                "pretty" | "toArray" => { a.max(0)?; }
                _ => return Err(self.err_at(call_pos, t!("core.mongo.parse.unsupported_method", name = method))),
            }
        }
        Ok(())
    }
}

fn build_op(a: &Args, coll: String) -> Result<MongoOp, ParseError> {
    Ok(match a.method {
        "find" => {
            a.max(2)?;
            MongoOp::Find { coll, filter: a.doc_or_empty(0)?, projection: a.opt_doc(1)?, sort: None, skip: None, limit: None }
        }
        "findOne" => {
            a.max(2)?;
            MongoOp::FindOne { coll, filter: a.doc_or_empty(0)?, projection: a.opt_doc(1)? }
        }
        "countDocuments" => {
            a.max(1)?;
            MongoOp::Count { coll, filter: a.doc_or_empty(0)? }
        }
        "distinct" => {
            a.max(2)?;
            MongoOp::Distinct { coll, field: a.string(0)?, filter: a.doc_or_empty(1)? }
        }
        "aggregate" => {
            a.max(1)?;
            MongoOp::Aggregate { coll, pipeline: a.doc_array(0)? }
        }
        "getIndexes" => {
            a.max(0)?;
            MongoOp::GetIndexes { coll }
        }
        "insertOne" => {
            a.max(1)?;
            MongoOp::InsertOne { coll, doc: a.doc(0)? }
        }
        "insertMany" => {
            a.max(1)?;
            MongoOp::InsertMany { coll, docs: a.doc_array(0)? }
        }
        "updateOne" => {
            a.max(2)?;
            MongoOp::UpdateOne { coll, filter: a.doc(0)?, update: a.update(1)? }
        }
        "updateMany" => {
            a.max(2)?;
            MongoOp::UpdateMany { coll, filter: a.doc(0)?, update: a.update(1)? }
        }
        "replaceOne" => {
            a.max(2)?;
            MongoOp::ReplaceOne { coll, filter: a.doc(0)?, doc: a.doc(1)? }
        }
        "deleteOne" => {
            a.max(1)?;
            MongoOp::DeleteOne { coll, filter: a.doc(0)? }
        }
        "deleteMany" => {
            a.max(1)?;
            MongoOp::DeleteMany { coll, filter: a.doc(0)? }
        }
        other => return Err(a.err(a.call_pos, t!("core.mongo.parse.unsupported_method", name = other))),
    })
}
```

```rust
use std::collections::HashMap;

/// 一次查询的展示数据。`rows` 与关系库的 `QueryResult.rows` 同形,网格直接复用;
/// `docs` / `ids` 与 `rows` 按下标一一对应。
#[derive(Debug, Clone, Default, PartialEq)]
pub struct MongoResult {
    /// 顶层字段并集,按首次出现顺序,`_id` 固定第一列
    pub columns: Vec<String>,
    /// None = 该文档没有这个字段(与值为 null 区分)
    pub rows: Vec<Vec<Option<String>>>,
    /// 每篇文档的 mongosh 风格缩进文本(JSON 视图 / 文档编辑器)
    pub docs: Vec<String>,
    /// 原始 `_id`,编辑 / 删除时定位
    pub ids: Vec<Option<Bson>>,
    /// 结果被 max_rows 截断
    pub truncated: bool,
    /// 写操作或标量结果的摘要
    pub summary: Option<String>,
}

pub fn docs_to_result(docs: Vec<Document>, truncated: bool) -> MongoResult {
    let mut columns: Vec<String> = vec![];
    if docs.iter().any(|d| d.contains_key("_id")) {
        columns.push("_id".into());
    }
    for d in &docs {
        for k in d.keys() {
            if !columns.iter().any(|c| c == k) {
                columns.push(k.clone());
            }
        }
    }
    let rows = docs
        .iter()
        .map(|d| columns.iter().map(|c| d.get(c).map(cell_text)).collect())
        .collect();
    MongoResult {
        rows,
        docs: docs.iter().map(|d| to_shell(&Bson::Document(d.clone()), true)).collect(),
        ids: docs.iter().map(|d| d.get("_id").cloned()).collect(),
        columns,
        truncated,
        summary: None,
    }
}

/// 单列结果(distinct、计数)。
pub fn values_to_result(column: &str, values: Vec<Bson>) -> MongoResult {
    MongoResult {
        columns: vec![column.to_string()],
        rows: values.iter().map(|v| vec![Some(cell_text(v))]).collect(),
        docs: values.iter().map(|v| to_shell(v, true)).collect(),
        ids: vec![None; values.len()],
        truncated: false,
        summary: None,
    }
}

/// 列 → 网格配色用的类型名(取该列第一个非 null 值),名称对齐 `grid::classify_cell` 的关键字。
pub fn column_types(docs: &[Document]) -> HashMap<String, String> {
    let mut out = HashMap::new();
    for d in docs {
        for (k, v) in d {
            if out.contains_key(k) {
                continue;
            }
            let t = match v {
                Bson::Null => continue,
                Bson::ObjectId(_) | Bson::Binary(_) => "uuid",
                Bson::Int32(_) => "integer",
                Bson::Int64(_) => "bigint",
                Bson::Double(_) => "double",
                Bson::Decimal128(_) => "numeric",
                Bson::DateTime(_) | Bson::Timestamp(_) => "timestamp",
                Bson::Boolean(_) => "boolean",
                Bson::String(_) => "text",
                _ => "json",
            };
            out.insert(k.clone(), t.to_string());
        }
    }
    out
}
```

```rust
/// MongoDB 风险审查:基于解析后的语句,不做文本匹配。`src` 用来取每条语句的原文。
pub fn analyze_mongo(
    src: &str,
    stmts: &[crate::mongo::parse::Statement],
) -> Vec<RiskFinding> {
    use crate::mongo::parse::MongoOp;
    let mut out = vec![];
    for s in stmts {
        let (level, kind) = match &s.op {
            MongoOp::DeleteMany { filter, .. } if filter.is_empty() => (RiskLevel::Danger, "deleteMany({})"),
            MongoOp::UpdateMany { filter, .. } if filter.is_empty() => (RiskLevel::Danger, "updateMany({})"),
            MongoOp::DeleteMany { .. } => (RiskLevel::Warning, "deleteMany"),
            MongoOp::UpdateMany { .. } => (RiskLevel::Warning, "updateMany"),
            MongoOp::DeleteOne { filter, .. } if filter.is_empty() => (RiskLevel::Warning, "deleteOne({})"),
            MongoOp::UpdateOne { filter, .. } if filter.is_empty() => (RiskLevel::Warning, "updateOne({})"),
            MongoOp::ReplaceOne { filter, .. } if filter.is_empty() => (RiskLevel::Warning, "replaceOne({})"),
            MongoOp::Aggregate { pipeline, .. } if pipeline.iter().any(|st| st.contains_key("$out")) => {
                (RiskLevel::Danger, "aggregate $out")
            }
            MongoOp::Aggregate { pipeline, .. } if pipeline.iter().any(|st| st.contains_key("$merge")) => {
                (RiskLevel::Warning, "aggregate $merge")
            }
            _ => continue,
        };
        out.push(RiskFinding {
            sql: src[s.range.clone()].trim().to_string(),
            level,
            kind: kind.to_string(),
            object: s.op.collection().map(str::to_string),
        });
    }
    out
}
```

先读 `completion.rs` 里 `redis_completions` 的返回结构(`CompletionResult { replace: Range<usize>, items: Vec<…> }` 及 item 的字段名),按同样的结构返回:

```rust
/// mongosh 的补全:`db.` 之后补集合名,`db.<集合>.` 之后补方法名。
pub fn mongo_completions(
    text: &str,
    offset: usize,
    collections: &[String],
    explicit: bool,
) -> Option<CompletionResult> {
    const METHODS: [&str; 13] = [
        "find", "findOne", "countDocuments", "distinct", "aggregate", "getIndexes",
        "insertOne", "insertMany", "updateOne", "updateMany", "replaceOne", "deleteOne", "deleteMany",
    ];
    let before = &text[..offset.min(text.len())];
    let word_start = before
        .rfind(|c: char| !(c.is_alphanumeric() || c == '_' || c == '$' || c == '-'))
        .map_or(0, |i| i + 1);
    let word = &before[word_start..];
    let head = &before[..word_start];
    let lower = word.to_lowercase();
    let items: Vec<CompletionItem> = if head.ends_with("db.") && !head.ends_with(".db.") {
        collections
            .iter()
            .filter(|c| c.to_lowercase().starts_with(&lower))
            .map(|c| {
                let insert = crate::mongo::parse::coll_ref(c);
                CompletionItem {
                    label: c.clone(),
                    insert: insert.strip_prefix("db.").unwrap_or(&insert).to_string(),
                    detail: String::new(),
                    kind: ItemKind::Table,
                }
            })
            .collect()
    } else if head.ends_with('.') && head.trim_end_matches('.').rsplit_once("db.").is_some() {
        METHODS
            .iter()
            .filter(|m| m.to_lowercase().starts_with(&lower))
            .map(|m| CompletionItem {
                label: m.to_string(),
                insert: m.to_string(),
                detail: String::new(),
                kind: ItemKind::Command,
            })
            .collect()
    } else {
        return None;
    };
    if items.is_empty() && !explicit {
        return None;
    }
    Some(CompletionResult { replace: word_start..offset, items })
}
```

```rust
//! MongoDB 连接池:按连接 id 缓存 `mongodb::Client`(其内部自带连接池)及其当前库。

use super::convert::{MongoResult, docs_to_result, values_to_result};

    max_time: StdMutex<Duration>,

fn server_code(e: &mongodb::error::Error) -> Option<i32> {
    use mongodb::error::{ErrorKind as K, WriteFailure};
    match e.kind.as_ref() {
        K::Command(c) => Some(c.code),
        K::Write(WriteFailure::WriteError(w)) => Some(w.code),
        _ => None,
    }
}

/// 驱动错误 → AppError:常见错误码附本地化提示,detail 保留 `code codeName: 原文`。
pub(crate) fn merr(e: mongodb::error::Error) -> AppError {
    use mongodb::error::ErrorKind as K;
    let message = match (server_code(&e), e.kind.as_ref()) {
        (Some(13), _) => t!("core.mongo.err.unauthorized"),
        (Some(11000), _) => t!("core.mongo.err.duplicate_key"),
        (Some(50), _) => t!("core.mongo.err.max_time"),
        (_, K::ServerSelection { .. }) => t!("core.mongo.err.server_selection"),
        _ => t!("core.mongo.err.command_failed"),
    };
    let detail = match server_code(&e) {
        Some(code) => format!("{code} {e}"),
        None => e.to_string(),
    };
    AppError::new(ErrorKind::Mongo, message).with_detail(detail)
}

async fn open(uri: &str) -> AppResult<Client> {
    crate::tls::install_default_provider();
    let mut opts = ClientOptions::parse(uri).await.map_err(|e| {
        AppError::new(ErrorKind::Connection, t!("core.mongo.err.invalid_uri")).with_detail(e.to_string())
    })?;
    opts.app_name = Some("Datum".into());
    opts.server_selection_timeout = Some(Duration::from_secs(5));
    opts.connect_timeout = Some(Duration::from_secs(10));
    let client = Client::with_options(opts).map_err(merr)?;
    client
        .database("admin")
        .run_command(doc! { "ping": 1 })
        .await
        .map_err(|e| AppError::new(ErrorKind::Connection, t!("core.err.connect_failed", db = "MongoDB")).with_detail(merr(e).display()))?;
    Ok(client)
}

/// 依次尝试候选连接串(`Prefer` 有两个),返回第一个成功的;都失败时返回最后一个错误。
async fn open_first(uris: &[String]) -> AppResult<Client> {
    let mut last = AppError::new(ErrorKind::Connection, t!("core.mongo.err.invalid_uri"));
    for u in uris {
        match open(u).await {
            Ok(c) => return Ok(c),
            Err(e) => last = e,
        }
    }
    Err(last)
}

pub async fn test_connect(uris: &[String]) -> AppResult<()> {
    open_first(uris).await.map(|_| ())
}

async fn collect(mut cursor: Cursor<Document>, max: usize) -> AppResult<(Vec<Document>, bool)> {
    let mut out = vec![];
    while let Some(d) = cursor.try_next().await.map_err(merr)? {
        if out.len() == max {
            return Ok((out, true));
        }
        out.push(d);
    }
    Ok((out, false))
}

impl MongoPool {
    pub fn set_max_time(&self, d: Duration) {
        *self.max_time.lock().unwrap() = d;
    }
    fn max_time(&self) -> Duration {
        *self.max_time.lock().unwrap()
    }

    pub async fn connect(&self, id: &str, uris: &[String], db: String) -> AppResult<()> {
        let client = open_first(uris).await?;
        self.conns.lock().await.insert(id.to_string(), (client, db));
        Ok(())
    }

    pub async fn disconnect(&self, id: &str) {
        if let Some((c, _)) = self.conns.lock().await.remove(id) {
            c.shutdown().await;
        }
    }

    async fn handle(&self, id: &str) -> AppResult<(Client, String)> {
        self.conns
            .lock()
            .await
            .get(id)
            .cloned()
            .ok_or_else(|| AppError::new(ErrorKind::Connection, t!("core.err.not_connected")))
    }

    /// 当前库;未选库时报错(集合级操作都需要库)。
    async fn db(&self, id: &str) -> AppResult<mongodb::Database> {
        let (client, db) = self.handle(id).await?;
        if db.is_empty() {
            return Err(AppError::new(ErrorKind::Mongo, t!("core.mongo.err.no_database")));
        }
        Ok(client.database(&db))
    }

    /// 库列表:隐藏系统库(当前库除外)。无权列库(code 13)时退回只显示配置的库。
    pub async fn list_databases(&self, id: &str) -> AppResult<Vec<String>> {
        let (client, current) = self.handle(id).await?;
        match client.list_database_names().await {
            Ok(mut names) => {
                names.retain(|n| n == &current || !SYSTEM_DBS.contains(&n.as_str()));
                names.sort();
                Ok(names)
            }
            Err(e) if server_code(&e) == Some(13) => {
                if current.is_empty() {
                    Err(AppError::new(ErrorKind::Mongo, t!("core.mongo.err.cannot_list_dbs")))
                } else {
                    Ok(vec![current])
                }
            }
            Err(e) => Err(merr(e)),
        }
    }

    /// 当前库的集合(含视图),未选库时为空。
    pub async fn list_collections(&self, id: &str) -> AppResult<Vec<String>> {
        let (client, db) = self.handle(id).await?;
        if db.is_empty() {
            return Ok(vec![]);
        }
        let mut names = client.database(&db).list_collection_names().await.map_err(merr)?;
        names.sort();
        Ok(names)
    }

    pub async fn find_page(&self, id: &str, coll: &str, page: FindPage) -> AppResult<MongoResult> {
        let c = self.db(id).await?.collection::<Document>(coll);
        let mut q = c.find(page.filter).skip(page.skip).limit(page.limit).max_time(self.max_time());
        if let Some(s) = page.sort {
            q = q.sort(s);
        }
        if let Some(p) = page.projection {
            q = q.projection(p);
        }
        let (docs, _) = collect(q.await.map_err(merr)?, page.limit.max(0) as usize).await?;
        Ok(docs_to_result(docs, false))
    }

    /// 无过滤条件时用估算计数(快,不精确),有条件时精确计数。
    pub async fn count(&self, id: &str, coll: &str, filter: Option<Document>) -> AppResult<(u64, bool)> {
        let c = self.db(id).await?.collection::<Document>(coll);
        match filter {
            None => c.estimated_document_count().max_time(self.max_time()).await.map(|n| (n, false)).map_err(merr),
            Some(f) => c.count_documents(f).max_time(self.max_time()).await.map(|n| (n, true)).map_err(merr),
        }
    }

    pub async fn replace_one(&self, id: &str, coll: &str, doc_id: Bson, doc: Document) -> AppResult<()> {
        let c = self.db(id).await?.collection::<Document>(coll);
        let r = c.replace_one(doc! { "_id": doc_id }, doc).await.map_err(merr)?;
        if r.matched_count == 0 {
            return Err(AppError::new(ErrorKind::Mongo, t!("core.mongo.err.doc_gone")));
        }
        Ok(())
    }

    pub async fn insert_one(&self, id: &str, coll: &str, doc: Document) -> AppResult<Bson> {
        let c = self.db(id).await?.collection::<Document>(coll);
        Ok(c.insert_one(doc).await.map_err(merr)?.inserted_id)
    }

    pub async fn delete_one(&self, id: &str, coll: &str, doc_id: Bson) -> AppResult<()> {
        let c = self.db(id).await?.collection::<Document>(coll);
        let r = c.delete_one(doc! { "_id": doc_id }).await.map_err(merr)?;
        if r.deleted_count == 0 {
            return Err(AppError::new(ErrorKind::Mongo, t!("core.mongo.err.doc_gone")));
        }
        Ok(())
    }

    /// 依次执行;返回最后一个读操作的结果,所有语句的摘要合并到 `summary`。
    /// `read_only` 时含写操作的脚本整体拒绝(MongoDB 没有会话级只读,只能在客户端拦)。
    pub async fn execute(&self, id: &str, ops: Vec<MongoOp>, max_rows: u32, read_only: bool) -> AppResult<MongoResult> {
        if read_only && ops.iter().any(MongoOp::is_write) {
            return Err(AppError::new(ErrorKind::Mongo, t!("core.mongo.err.read_only")));
        }
        let max = max_rows as usize;
        let mut last_read: Option<MongoResult> = None;
        let mut summaries = vec![];
        for (i, op) in ops.into_iter().enumerate() {
            let step = self.execute_one(id, op, max).await.map_err(|e| {
                let msg = t!("core.mongo.err.statement_failed", n = i + 1, message = e.message).to_string();
                AppError { message: msg, ..e }
            })?;
            match step {
                Step::Read(r) => last_read = Some(r),
                Step::Summary(s) => summaries.push(s),
            }
        }
        let mut out = last_read.unwrap_or_default();
        if !summaries.is_empty() {
            out.summary = Some(summaries.join("\n"));
        }
        Ok(out)
    }

    async fn execute_one(&self, id: &str, op: MongoOp, max: usize) -> AppResult<Step> {
        let mt = self.max_time();
        if let MongoOp::ShowDbs = op {
            let names = self.list_databases(id).await?;
            return Ok(Step::Read(values_to_result("name", names.into_iter().map(Bson::String).collect())));
        }
        let db = self.db(id).await?;
        let coll = |name: &str| db.collection::<Document>(name);
        Ok(match op {
            MongoOp::ShowDbs => unreachable!(),
            MongoOp::ShowCollections => {
                let mut names = db.list_collection_names().await.map_err(merr)?;
                names.sort();
                Step::Read(values_to_result("name", names.into_iter().map(Bson::String).collect()))
            }
            MongoOp::Find { coll: c, filter, projection, sort, skip, limit } => {
                let cap = limit.filter(|l| *l > 0).map_or(max, |l| (l as usize).min(max));
                let mut q = coll(&c).find(filter).max_time(mt).limit(cap as i64 + 1);
                if let Some(p) = projection { q = q.projection(p); }
                if let Some(s) = sort { q = q.sort(s); }
                if let Some(s) = skip { q = q.skip(s); }
                let (docs, truncated) = collect(q.await.map_err(merr)?, cap).await?;
                // 用户自己写的 limit 正好取满不算截断
                Step::Read(docs_to_result(docs, truncated && cap == max))
            }
            MongoOp::FindOne { coll: c, filter, projection } => {
                let mut q = coll(&c).find_one(filter).max_time(mt);
                if let Some(p) = projection { q = q.projection(p); }
                Step::Read(docs_to_result(q.await.map_err(merr)?.into_iter().collect(), false))
            }
            MongoOp::Count { coll: c, filter } => {
                let n = coll(&c).count_documents(filter).max_time(mt).await.map_err(merr)?;
                Step::Read(values_to_result("count", vec![Bson::Int64(n as i64)]))
            }
            MongoOp::Distinct { coll: c, field, filter } => {
                let vals = coll(&c).distinct(&field, filter).max_time(mt).await.map_err(merr)?;
                Step::Read(values_to_result(&field, vals))
            }
            MongoOp::Aggregate { coll: c, pipeline } => {
                let writes = pipeline.iter().any(|s| s.contains_key("$out") || s.contains_key("$merge"));
                let cursor = coll(&c).aggregate(pipeline).max_time(mt).await.map_err(merr)?;
                let (docs, truncated) = collect(cursor, max).await?;
                if writes {
                    Step::Summary(t!("core.mongo.summary.aggregate_written").to_string())
                } else {
                    Step::Read(docs_to_result(docs, truncated))
                }
            }
            MongoOp::GetIndexes { coll: c } => {
                let cursor = coll(&c).list_indexes().await.map_err(merr)?;
                let idx: Vec<mongodb::IndexModel> = cursor.try_collect().await.map_err(merr)?;
                let docs = idx
                    .into_iter()
                    .map(|m| mongodb::bson::to_document(&m).unwrap_or_default())
                    .collect();
                Step::Read(docs_to_result(docs, false))
            }
            MongoOp::InsertOne { coll: c, doc } => {
                coll(&c).insert_one(doc).await.map_err(merr)?;
                Step::Summary(t!("core.mongo.summary.inserted", n = 1).to_string())
            }
            MongoOp::InsertMany { coll: c, docs } => {
                let r = coll(&c).insert_many(docs).await.map_err(merr)?;
                Step::Summary(t!("core.mongo.summary.inserted", n = r.inserted_ids.len()).to_string())
            }
            MongoOp::UpdateOne { coll: c, filter, update } => {
                let r = coll(&c).update_one(filter, to_modifications(update)).await.map_err(merr)?;
                Step::Summary(t!("core.mongo.summary.updated", matched = r.matched_count, modified = r.modified_count).to_string())
            }
            MongoOp::UpdateMany { coll: c, filter, update } => {
                let r = coll(&c).update_many(filter, to_modifications(update)).await.map_err(merr)?;
                Step::Summary(t!("core.mongo.summary.updated", matched = r.matched_count, modified = r.modified_count).to_string())
            }
            MongoOp::ReplaceOne { coll: c, filter, doc } => {
                let r = coll(&c).replace_one(filter, doc).await.map_err(merr)?;
                Step::Summary(t!("core.mongo.summary.updated", matched = r.matched_count, modified = r.modified_count).to_string())
            }
            MongoOp::DeleteOne { coll: c, filter } => {
                let r = coll(&c).delete_one(filter).await.map_err(merr)?;
                Step::Summary(t!("core.mongo.summary.deleted", n = r.deleted_count).to_string())
            }
            MongoOp::DeleteMany { coll: c, filter } => {
                let r = coll(&c).delete_many(filter).await.map_err(merr)?;
                Step::Summary(t!("core.mongo.summary.deleted", n = r.deleted_count).to_string())
            }
        })
    }
}

enum Step {
    Read(MongoResult),
    Summary(String),
}

fn to_modifications(u: Update) -> mongodb::options::UpdateModifications {
    match u {
        Update::Doc(d) => mongodb::options::UpdateModifications::Document(d),
        Update::Pipeline(p) => mongodb::options::UpdateModifications::Pipeline(p),
    }
}
```

```rust
//! MongoDB 的 tab 逻辑:集合浏览(过滤栏 + 服务端分页)、查询执行与文档操作。
//! 结果仍以 `QueryResult` 形态交给网格;`docs` / `ids` 与网格行按下标对应。

use super::mongo_filter::FilterBar;
use super::{TabEvent, WorkTab};
use crate::runtime;
use crate::store::Toast;
use datum_core::mongo::convert::MongoResult;
use datum_core::mongo::parse::parse_document;
use datum_core::pg::client::QueryResult;
use datum_core::service::{self, FindPage};
use gpui_kit::{AsyncWindowContext, Context, Subscription, Window};
use mongodb::bson::{Bson, Document};
use rust_i18n::t;
use std::collections::HashMap;
use std::sync::Arc;
use std::time::Instant;

pub const PAGE_SIZES: [usize; 3] = [50, 100, 500];

#[derive(Default)]
pub struct MongoTabState {
    /// 集合 tab 的集合名;查询 tab 为 None
    pub coll: Option<String>,
    pub bar: Option<FilterBar>,
    pub page: usize,
    pub page_size: usize,
    /// (数量, 是否精确)
    pub total: Option<(u64, bool)>,
    pub docs: Arc<Vec<String>>,
    pub ids: Arc<Vec<Option<Bson>>>,
    pub types: HashMap<String, String>,
    pub json_view: bool,
    pub truncated: bool,
    pub summary: Option<String>,
    pub bar_error: Option<String>,

pub fn permute<T: Clone>(v: &[T], order: &[usize]) -> Vec<T> {
    order.iter().filter_map(|&i| v.get(i).cloned()).collect()
}

/// 新建 tab 时的 Mongo 状态;集合 tab 带过滤栏。
pub(super) fn init(
    coll: Option<&str>,
    window: &mut Window,
    cx: &mut Context<WorkTab>,
) -> (MongoTabState, Vec<Subscription>) {
    let (bar, subs) = match coll {
        Some(_) => {
            let (bar, subs) = FilterBar::new(window, cx);
            (Some(bar), subs)
        }
        None => (None, vec![]),
    };
    (
        MongoTabState { coll: coll.map(str::to_string), bar, page_size: 100, ..Default::default() },
        subs,
    )
}

impl WorkTab {
    /// 视图行 → (文档文本, `_id`)。
    pub fn mongo_doc_at(&self, view_ix: usize) -> Option<(String, Option<Bson>)> {
        let m = self.mongo.as_ref()?;
        let ri = *self.view.get(view_ix)?;
        Some((m.docs.get(ri)?.clone(), m.ids.get(ri)?.clone()))
    }

    pub(super) fn set_mongo_result(&mut self, r: MongoResult, cx: &mut Context<Self>) {
        let MongoResult { columns, rows, docs, ids, truncated, summary } = r;
        let types = docs
            .iter()
            .filter_map(|d| parse_document(d).ok())
            .take(50)
            .collect::<Vec<_>>();
        if let Some(m) = self.mongo.as_mut() {
            m.types = datum_core::mongo::convert::column_types(&types);
            m.docs = Arc::new(docs);
            m.ids = Arc::new(ids);
            m.truncated = truncated;
            m.summary = summary;
        }
        self.result_is_query = !columns.is_empty() || self.mongo.as_ref().is_some_and(|m| m.summary.is_none());
        self.set_result(QueryResult { columns, rows, affected: None }, cx);
    }

    /// 解析过滤栏;失败时把错误显示在栏下方并返回 None。
    fn mongo_read_bar(&mut self, cx: &mut Context<Self>) -> Option<(Document, Option<Document>, Option<Document>)> {
        let m = self.mongo.as_mut()?;
        let bar = m.bar.as_ref()?;
        let read = |e: &gpui_kit::Entity<gpui_kit::component::input::InputState>| e.read(cx).value().to_string();
        let (f, s, p) = (read(&bar.filter), read(&bar.sort), read(&bar.projection));
        let parse = |label: &str, src: &str| {
            parse_document(src).map_err(|e| format!("{label}: {}", e.display()))
        };
        let result = (|| {
            let filter = parse("Filter", &f)?;
            let sort = parse("Sort", &s)?;
            let proj = parse("Projection", &p)?;
            Ok::<_, String>((filter, (!sort.is_empty()).then_some(sort), (!proj.is_empty()).then_some(proj)))
        })();
        match result {
            Ok(v) => {
                m.bar_error = None;
                Some(v)
            }
            Err(e) => {
                m.bar_error = Some(e);
                cx.notify();
                None
            }
        }
    }

    pub fn mongo_apply_filter(&mut self, window: &mut Window, cx: &mut Context<Self>) {
        if let Some(m) = self.mongo.as_mut() {
            m.total = None;
        }
        self.mongo_goto_page(0, window, cx);
    }

    pub fn mongo_reset_filter(&mut self, window: &mut Window, cx: &mut Context<Self>) {
        if let Some(bar) = self.mongo.as_ref().and_then(|m| m.bar.as_ref()) {
            for e in [&bar.filter, &bar.sort, &bar.projection] {
                e.update(cx, |s, cx| s.set_value("", window, cx));
            }
        }
        self.mongo_apply_filter(window, cx);
    }

    pub(super) fn mongo_refresh(&mut self, window: &mut Window, cx: &mut Context<Self>) {
        let page = self.mongo.as_ref().map_or(0, |m| m.page);
        self.mongo_goto_page(page, window, cx);
    }

    pub fn mongo_goto_page(&mut self, page: usize, window: &mut Window, cx: &mut Context<Self>) {
        let Some(coll) = self.mongo.as_ref().and_then(|m| m.coll.clone()) else { return };
        let Some((filter, sort, projection)) = self.mongo_read_bar(cx) else { return };
        let page_size = self.mongo.as_ref().map_or(100, |m| m.page_size);
        let need_total = self.mongo.as_ref().is_some_and(|m| m.total.is_none()) && filter.is_empty();
        let id = self.conn_id.clone();
        let req = FindPage { filter, sort, projection, skip: (page * page_size) as u64, limit: page_size as i64 };
        self.running = true;
        cx.notify();
        let started = Instant::now();
        self._task = Some(cx.spawn_in(window, async move |this, cx: &mut AsyncWindowContext| {
            let result = runtime::backend(move |s| async move {
                let r = service::mongo_find_page(&s, &id, &coll, req).await?;
                let total = if need_total { Some(service::mongo_count(&s, &id, &coll, None).await?) } else { None };
                Ok::<_, datum_core::error::AppError>((r, total))
            })
            .await;
            let elapsed = started.elapsed().as_millis();
            let _ = this.update(cx, |this, cx| {
                this.running = false;
                match result {
                    Ok((r, total)) => {
                        this.err = None;
                        this.elapsed_ms = Some(elapsed);
                        if let Some(m) = this.mongo.as_mut() {
                            m.page = page;
                            if total.is_some() {
                                m.total = total;
                            }
                        }
                        this.set_mongo_result(r, cx);
                    }
                    Err(e) => {
                        this.err = Some(e.display());
                        cx.notify();
                    }
                }
            });
        }));
    }

    /// 有过滤条件时手动精确计数(大集合上可能很慢,所以不自动执行)。
    pub fn mongo_count(&mut self, window: &mut Window, cx: &mut Context<Self>) {
        let Some(coll) = self.mongo.as_ref().and_then(|m| m.coll.clone()) else { return };
        let Some((filter, _, _)) = self.mongo_read_bar(cx) else { return };
        let id = self.conn_id.clone();
        cx.spawn_in(window, async move |this, cx: &mut AsyncWindowContext| {
            let r = runtime::backend(move |s| async move {
                service::mongo_count(&s, &id, &coll, Some(filter)).await
            })
            .await;
            let _ = this.update(cx, |this, cx| match r {
                Ok(n) => {
                    if let Some(m) = this.mongo.as_mut() {
                        m.total = Some(n);
                    }
                    cx.notify();
                }
                Err(e) => cx.emit(TabEvent::Toast(Toast::Error(e.display().into()))),
            });
        })
        .detach();
    }

    /// 点表头:集合 tab 写入 Sort 栏并服务端重查(倒序 → 升序 → 无);查询 tab 客户端排序,
    /// 行、文档文本、`_id` 一起重排,保证编辑 / 删除作用在点中的那一篇上。
    pub(super) fn mongo_cycle_sort(&mut self, col: &str, window: &mut Window, cx: &mut Context<Self>) {
        let next = match &self.sort {
            Some((c, true)) if c == col => Some((col.to_string(), false)),
            Some((c, false)) if c == col => None,
            _ => Some((col.to_string(), true)),
        };
        if let Some(bar) = self.mongo.as_ref().and_then(|m| m.bar.as_ref()) {
            let text = match &next {
                Some((c, desc)) => format!("{{ {}: {} }}", datum_core::mongo::convert::to_shell(&Bson::String(c.clone()), false), if *desc { -1 } else { 1 }),
                None => String::new(),
            };
            bar.sort.update(cx, |s, cx| s.set_value(text, window, cx));
            self.sort = next;
            self.mongo_goto_page(0, window, cx);
            return;
        }
        let (Some(result), Some(m)) = (self.result.clone(), self.mongo.as_mut()) else { return };
        let Some(ci) = result.columns.iter().position(|c| c == col) else { return };
        let desc = matches!(&self.sort, Some((c, false)) if c == col);
        let order = datum_core::grid::sort_order(&result.rows, ci, desc);
        m.docs = Arc::new(permute(&m.docs, &order));
        m.ids = Arc::new(permute(&m.ids, &order));
        self.sort = Some((col.to_string(), desc));
        self.set_result(
            QueryResult { columns: result.columns.clone(), rows: permute(&result.rows, &order), affected: None },
            cx,
        );
    }
}
```

注意 `mongo_cycle_sort` 里写入 Sort 栏的 key 用 `to_shell(String)` 生成带引号的字段名(`{ "a.b": -1 }`),这样嵌套字段名也能被正确解析。

```rust
//! 保存失败时弹窗保持打开,编辑内容不丢。

use crate::runtime;
use crate::store::Toast;
use crate::tab::WorkTab;
use crate::theme::ActivePalette as _;
use datum_core::mongo::parse::parse_document;
use datum_core::service;

use gpui_kit::component::{ActiveTheme as _, WindowExt as _, h_flex, v_flex};

    Styled as _, Subscription, Task, WeakEntity, Window, div, px,

use mongodb::bson::{Bson, Document};

pub struct DocTarget {
    pub tab: WeakEntity<WorkTab>,
    pub conn_id: String,
    pub coll: String,

/// 编辑后的文档 → 用于 replaceOne 的文档:删掉 `_id` 时补回原值;改了 `_id`(含改类型)时拒绝。
pub fn prepare_replacement(original_id: &Bson, mut doc: Document) -> Result<Document, String> {
    match doc.get("_id") {
        None => {
            let mut out = Document::new();
            out.insert("_id", original_id.clone());
            out.extend(doc);
            Ok(out)
        }
        Some(id) if id == original_id => {
            doc.insert("_id", original_id.clone());
            Ok(doc)
        }
        Some(_) => Err(t!("mongo.id_changed").to_string()),
    }
}

struct DocEditor {
    target: DocTarget,
    original_id: Option<Bson>,
    editor: Entity<EditorState>,
    error: Option<String>,
    saving: bool,
    _task: Option<Task<()>>,
    _sub: Subscription,
}

impl DocEditor {
    fn parsed(&self, cx: &gpui_kit::App) -> Result<Document, String> {
        parse_document(&self.editor.read(cx).value()).map_err(|e| e.display())
    }

    fn save(&mut self, window: &mut Window, cx: &mut Context<Self>) {
        if self.saving {
            return;
        }
        let doc = match self.parsed(cx) {
            Ok(d) => d,
            Err(e) => {
                self.error = Some(e);
                cx.notify();
                return;
            }
        };
        let doc = match &self.original_id {
            Some(id) => match prepare_replacement(id, doc) {
                Ok(d) => d,
                Err(e) => {
                    self.error = Some(e);
                    cx.notify();
                    return;
                }
            },
            None => doc,
        };
        let (id, coll, original) = (self.target.conn_id.clone(), self.target.coll.clone(), self.original_id.clone());
        let tab = self.target.tab.clone();
        self.saving = true;
        cx.notify();
        self._task = Some(cx.spawn_in(window, async move |this, cx: &mut AsyncWindowContext| {
            let r = runtime::backend(move |s| async move {
                match original {
                    Some(doc_id) => service::mongo_replace_one(&s, &id, &coll, doc_id, doc).await,
                    None => service::mongo_insert_one(&s, &id, &coll, doc).await.map(|_| ()),
                }
            })
            .await;
            let _ = this.update_in(cx, |this, window, cx| {
                this.saving = false;
                match r {
                    Ok(()) => {
                        window.close_dialog(cx);
                        let _ = tab.update(cx, |t, cx| {
                            cx.emit(crate::tab::TabEvent::Toast(Toast::Success(t!("mongo.saved").to_string().into())));
                            t.refresh(window, cx);
                        });
                    }
                    Err(e) => {
                        this.error = Some(e.display());
                        cx.notify();
                    }
                }
            });
        }));
    }
}

impl Render for DocEditor {
    fn render(&mut self, _: &mut Window, cx: &mut Context<Self>) -> impl IntoElement {
        let parse_err = self.parsed(cx).err();
        let shown_err = self.error.clone().or(parse_err.clone());
        v_flex()
            .w_full()
            .h(px(560.))
            .gap_2()
            .child(
                div()
                    .flex_1()
                    .min_h_0()
                    .rounded(cx.theme().radius)
```

```rust
    pub fn confirm_delete_doc(&mut self, view_ix: usize, window: &mut Window, cx: &mut Context<Self>) {

                database: self.connection(cx).map(|c| c.database).filter(|d| !d.is_empty()),
                dialect: None,
            },

                let (id, coll, doc_id, this) = (id.clone(), coll.clone(), doc_id.clone(), this.clone());

            },
```

- [ ] **Step 3: `run_mongo`(追加到 `tab/mongo.rs`)**

```rust
impl WorkTab {
    pub(super) fn run_mongo(
        &mut self,
        src: String,
        skip_history: bool,
        skip_risk: bool,
        window: &mut Window,
        cx: &mut Context<Self>,
    ) {
        let stmts = match datum_core::mongo::parse::parse_script(&src) {
            Ok(v) if v.is_empty() => {
                self.err = Some(t!("tab.empty_command").to_string());
                cx.notify();
                return;
            }
            Ok(v) => v,
            Err(e) => {
                self.err = Some(e.display());
                self.result = None;
                self.recompute_view(cx);
                return;
            }
        };
        if !skip_risk {
            let findings = datum_core::sql::risk::analyze_mongo(&src, &stmts);
            if !findings.is_empty() {
                self.read_only_next = false;
                let conn = self.connection(cx);
                let this = cx.entity().downgrade();
                let again = src.clone();
                crate::dialogs::risk::confirm_destructive(
                    crate::dialogs::risk::RiskGate {
                        findings,
                        target_label: conn.as_ref().map(|c| c.name.clone()).unwrap_or_default(),
                        database: conn.map(|c| c.database).filter(|d| !d.is_empty()),
                        dialect: None,
                    },
                    move |window, cx| {
                        let src = again.clone();
                        let _ = this.update(cx, |t, cx| t.run_mongo(src, skip_history, true, window, cx));
                    },
                    window,
                    cx,
                );
                return;
            }
        }
        let ops: Vec<_> = stmts.into_iter().map(|s| s.op).collect();
        let wrote = ops.iter().any(|o| o.is_write());
        let read_only = std::mem::take(&mut self.read_only_next);
        let (id, max_rows) = (self.conn_id.clone(), Self::max_rows(cx));
        self.running = true;
        cx.notify();
        let started = Instant::now();
        self._task = Some(cx.spawn_in(window, async move |this, cx: &mut AsyncWindowContext| {
            let r = runtime::backend(move |s| async move {
                service::mongo_execute(&s, &id, ops, max_rows, read_only).await
            })
            .await;
            let elapsed = started.elapsed().as_millis();
            let _ = this.update(cx, |this, cx| {
                this.running = false;
                match r {
                    Ok(res) => {
                        this.err = None;
                        this.elapsed_ms = Some(elapsed);
                        this.sort = None;
                        this.filter.reset_all();
                        let rows = res.rows.len() as u64;
                        let has_rows = !res.columns.is_empty();
                        this.set_mongo_result(res, cx);
                        if wrote {
                            cx.emit(TabEvent::SchemaChanged);
                        }
                        if !skip_history {
                            this.record_history(&src, elapsed, has_rows.then_some(rows), None, cx);
                        }
                    }
                    Err(e) => {
                        this.err = Some(e.display());
                        this.result = None;
                        this.recompute_view(cx);
                    }
                }
                cx.notify();
            });
        }));
    }
}
```

(`max_rows`、`record_history`、`connection` 是 `tab/mod.rs` 的私有项,子模块可访问。`insertOne` 等写操作可能新建集合,所以发 `SchemaChanged` 让侧边栏刷新集合列表。)
